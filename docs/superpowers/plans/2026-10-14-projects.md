# Projects Implementation Plan

> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** Add offline-only linked Projects (deadline, target hours, linked events, progress ring, manual complete/archive) without a 6th tab.

**Architecture:** New `projects` table + nullable `events.projectId` FK (schema v6); pure `ProjectProgress` math recomputed from events + sessions; new `lib/features/projects/` hub reusing calendar streams; event-sheet picker row; Profile card entry point.

**Tech Stack:** Flutter 3.44.7 / Dart 3.12.2, Riverpod 2.6.1, Drift 2.34.3 + drift_flutter 0.3.1, CustomPainter rings, flutter_test.

## Global Constraints

- Offline-only, on-device. No account, no backend, no sync, no HTTP client.
- Tab count stays 5. Do not modify `AppTabBar.tabCount` or add a 6th tab.
- Features never import each other: features may depend one-way toward projects via providers/sheets only, no cycles. `features/projects/` imports only `data/`, `core/`, `app/`, `shared/`.
- No hard-coded colours in feature code. All colors via `app/theme/tone_colors.dart` / `app/theme/tokens.dart` (the no-hardcoded-colour test must pass).
- All user-facing copy lives in `lib/core/strings.dart`, English-only.
- Tone always encodes Focus/Life/Inbox. Project tone-affinity is display-only and never rewrites event tones. Flight-forces-Inbox still wins.
- Past days are read-only. Picker and project edits disabled for past days.
- Progress is derived, never accumulated. A full recompute from source rows must always be possible.
- Every schema bump needs a migration step in `lib/data/database.dart` and a test in `test/data/migration_test.dart`; dumps live in `drift_schemas/`.
- Codegen after table changes runs with the Flutter SDK's `dart`: `"$(dirname "$(which flutter)")/dart" run build_runner build --delete-conflicting-outputs`.
- Repo gates: `flutter analyze`, `flutter test`. No golden regen expected (default dial pixels unchanged).

```dart
// lib/features/projects/model/project_progress.dart
abstract final class ProjectProgress {
  /// 0.0+ (may exceed 1.0; callers render overflow). Target mode wins when set.
  static double compute({required int plannedMinutes, required int actualMinutes, required int? targetMinutes}) {
    if (targetMinutes != null && targetMinutes > 0) return actualMinutes / targetMinutes;
    if (plannedMinutes <= 0) return 0.0;
    return actualMinutes / plannedMinutes;
  }

  static bool isReadyToComplete(double progress) => progress >= 1.0;

  static bool isOverdue(DateTime? deadline, DateTime now) =>
      deadline != null && _day(deadline).isBefore(_day(now));

  static DateTime _day(DateTime d) => DateTime(d.year, d.month, d.day);
}

/// 'Due Fri' / 'Due in 3d' / 'Overdue 3d' / '' when null. Pure of now for tests.
String deadlineLabel(DateTime? deadline, DateTime now) {
  if (deadline == null) return '';
  final a = DateTime(now.year, now.month, now.day);
  final b = DateTime(deadline.year, deadline.month, deadline.day);
  final diff = b.difference(a).inDays;
  if (diff < 0) return 'Overdue ${-diff}d';
  if (diff == 0) return 'Due today';
  if (diff <= 7) return 'Due in ${diff}d';
  const weekdays = ['Mon', 'Tue', 'Wed', 'Thu', 'Fri', 'Sat', 'Sun'];
  return 'Due ${weekdays[(b.weekday - 1) % 7]}';
}

String? validateProjectTitle(String value) {
  final v = value.trim();
  if (v.isEmpty) return 'Name your project';
  if (v.length > 80) return 'Keep it under 80 characters';
  return null;
}

String? validateTargetMinutes(int? value) {
  if (value == null) return null;
  if (value < 15) return 'Minimum 15 minutes';
  if (value > 600000) return 'Maximum 10,000 hours';
  return null;
}
```

```dart
// lib/data/daos/project_dao.dart
import 'package:drift/drift.dart';
import 'package:energy_clock/data/database.dart';
import 'package:energy_clock/data/tables/projects_table.dart';

part 'project_dao.g.dart';

@DriftAccessor(tables: [Projects, Events])
class ProjectDao extends DatabaseAccessor<AppDatabase> with _$ProjectDaoMixin {
  ProjectDao(super.attachedDatabase);

  Future<int> createProject(ProjectsCompanion entry) => into(projects).insert(entry);
  Future<List<ProjectRow>> allProjects() =>
      (select(projects)..orderBy([(r) => OrderingTerm(expression: r.createdAt, mode: OrderingMode.desc)]).limit(200)).get();
  Stream<List<ProjectRow>> watchProjects() =>
      (select(projects)..orderBy([(r) => OrderingTerm(expression: r.createdAt, mode: OrderingMode.desc)]).limit(200)).watch();
  Future<void> updateDeadline(int id, DateTime? deadline) =>
      (update(projects)..where((r) => r.id.equals(id))).write(ProjectsCompanion(deadline: Value(deadline)));
  Future<void> updateTarget(int id, int? targetMinutes) =>
      (update(projects)..where((r) => r.id.equals(id))).write(ProjectsCompanion(targetMinutes: Value(targetMinutes)));
  Future<void> updateAffinity(int id, int? toneIndex) =>
      (update(projects)..where((r) => r.id.equals(id))).write(ProjectsCompanion(toneIndex: Value(toneIndex)));
  Future<void> setStatus(int id, int status, DateTime? completedAt) =>
      (update(projects)..where((r) => r.id.equals(id))).write(
        ProjectsCompanion(status: Value(status), completedAt: Value(completedAt)),
      );
  /// Deletes the project row only; linked events keep their rows with projectId cleared.
  Future<void> deleteAndUnlink(int id) => transaction(() async {
        await (update(events)..where((r) => r.projectId.equals(id))).write(const EventsCompanion(projectId: Value(null)));
        await (delete(projects)..where((r) => r.id.equals(id))).go();
      });
}
```

```dart
// lib/data/repositories/project_repository.dart
import 'package:energy_clock/data/database.dart';

abstract interface class ProjectRepository {
  Stream<List<ProjectRow>> watchProjects();
  Future<int> create(String title);
  Future<void> complete(int id);
  Future<void> reopen(int id);
  Future<void> archive(int id);
  Future<void> deleteAndUnlink(int id);
}

class DriftProjectRepository implements ProjectRepository {
  DriftProjectRepository(this._db);
  final AppDatabase _db;

  @override
  Stream<List<ProjectRow>> watchProjects() => _db.projectDao.watchProjects();

  @override
  Future<int> create(String title) => _db.projectDao.createProject(
        ProjectsCompanion.insert(title: title.trim(), createdAt: DateTime.now()),
      );

  @override
  Future<void> complete(int id) =>
      _db.projectDao.setStatus(id, 1, DateTime.now());

  @override
  Future<void> reopen(int id) => _db.projectDao.setStatus(id, 0, null);

  @override
  Future<void> archive(int id) => _db.projectDao.setStatus(id, 2, null);

  @override
  Future<void> deleteAndUnlink(int id) => _db.projectDao.deleteAndUnlink(id);
}
```

```dart
              onTap: () => Navigator.pop(context, null),

              },
```
